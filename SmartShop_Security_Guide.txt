Fix #2 — Data Ownership Check
এখন যেকেউ যেকোনো order ID দিয়ে অন্যের order দেখতে পারে। প্রতিটি query তে shop filter যোগ করতে হবে।

models/Order.js এ shop field যোগ করো:
// Order schema তে এই line যোগ করো
shop: { type: mongoose.Schema.Types.ObjectId, ref: 'Shop', required: true },

controllers/orderController.js — সব query তে shop filter দাও:
// GET /api/orders
exports.getOrders = async (req, res, next) => {
  try {
    const filter = { shop: req.user.shopId }  // <-- এটা যোগ করো
    if (req.query.status) filter.status = req.query.status

    const orders = await Order.find(filter).populate('customer').sort({ createdAt: -1 })
    res.json(orders)
  } catch (err) { next(err) }
}

// GET /api/orders/:id — ownership check
exports.getOrder = async (req, res, next) => {
  try {
    const order = await Order.findOne({
      _id: req.params.id,
      shop: req.user.shopId   // <-- অন্যের order দেখতে পাবে না
    }).populate('customer')

    if (!order) return res.status(404).json({ message: 'Order পাওয়া যায়নি' })
    res.json(order)
  } catch (err) { next(err) }
}

// POST /api/orders — shop যোগ করো

  try {
    // ... existing code ...

      // ... existing fields ...
      shop: req.user.shopId,  // <-- এটা যোগ করো
    })
    // ...
  } catch (err) { next(err) }
}

Fix #3 — Input Validation
নতুন order তৈরির সময় কেউ ভুল data দিলে সেটা block করতে হবে।

routes/orderRoutes.js আপডেট করো:
const { body, validationResult } = require('express-validator')

// Validation rules
const orderValidation = [
  body('customerName').trim().notEmpty().withMessage('নাম দিতে হবে'),
  body('phone')
    .trim()
    .matches(/^01[3-9][0-9]{8}$/)
    .withMessage('সঠিক বাংলাদেশি নম্বর দাও'),
  body('totalAmount')
    .isFloat({ min: 1 })
    .withMessage('Amount অবশ্যই positive হতে হবে'),
  body('paymentMethod')
    .isIn(['COD', 'bKash', 'Nagad', 'Rocket', 'Partial'])
    .withMessage('সঠিক payment method দাও'),
]

// Middleware: validation error check করে
function validate(req, res, next) {
  const errors = validationResult(req)
  if (!errors.isEmpty()) {
    return res.status(400).json({ errors: errors.array() })
  }
  next()
}

Fix #4 — JWT httpOnly Cookie
JWT token localStorage এ রাখলে XSS attack এ চুরি যেতে পারে। httpOnly cookie অনেক বেশি safe।

controllers/authController.js — token cookie তে পাঠাও:
// Login এর response এ token cookie তে পাঠাও
function sendToken(res, user, statusCode) {
  const token = jwt.sign({ id: user._id, shopId: user.shopId }, process.env.JWT_SECRET, {
    expiresIn: '30d'
  })

  res.cookie('token', token, {
    httpOnly: true,    // JS দিয়ে access করা যাবে না
    secure: process.env.NODE_ENV === 'production',  // HTTPS only
    sameSite: 'strict',
    maxAge: 30 * 24 * 60 * 60 * 1000,  // 30 days
  })

  res.status(statusCode).json({
    user: { id: user._id, name: user.name, email: user.email }
    // token আর response body তে পাঠাবে না
  })
}

middleware/authMiddleware.js — cookie থেকে token নাও:
function protect(req, res, next) {
  // Cookie থেকে নাও (httpOnly)
  const token = req.cookies.token

  // Fallback: পুরনো Bearer token এখনো support করো
  || (req.headers.authorization?.startsWith('Bearer ')
    ? req.headers.authorization.split(' ')[1]
    : null)

  if (!token) return res.status(401).json({ message: 'Login করো' })

  try {
    req.user = jwt.verify(token, process.env.JWT_SECRET)
    next()
  } catch {
    res.status(401).json({ message: 'Token expired, আবার login করো' })
  }
}

Fix #5 — File Upload Validation
Logo upload এ কোনো check নেই। কেউ চাইলে .exe বা বড় file দিতে পারবে।

const multer = require('multer')
const path   = require('path')

const storage = multer.diskStorage({
  destination: 'uploads/',
  filename: (req, file, cb) => {
    // Original filename এর বদলে random name দাও
    cb(null, Date.now() + path.extname(file.originalname))
  },
})

const upload = multer({
  storage,
  limits: { fileSize: 2 * 1024 * 1024 },  // Max 2MB
  fileFilter: (req, file, cb) => {
    const allowed = /jpeg|jpg|png|webp/
    const isValid = allowed.test(file.mimetype) && allowed.test(path.extname(file.originalname).toLowerCase())
    if (isValid) cb(null, true)
    else cb(new Error('শুধু JPG, PNG, WebP file upload করা যাবে'))
  },
})

module.exports = upload
